Deployment Code (GitHub Pages)

This GitHub Actions workflow automatically builds your site and deploys it to GitHub Pages every time you push to main.

File name:
deploy.yml
File location:
.github/workflows/deploy.yml

Copy and paste this code into the file:

name: Deploy to GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - run: npm install
      - run: npm run build -- --base=./
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist
      - uses: actions/deploy-pages@v4
